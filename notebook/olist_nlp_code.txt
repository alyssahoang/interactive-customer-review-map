# %% [markdown]
# #### 1.2.4 Text-length threshold
# 
# We compare candidate minimum-length rules and retain a threshold that balances quality and sample size.
# 

# %%
threshold_rows = []
text_all = raw_reviews["review_text"].fillna("").astype(str)
non_empty_base = int((text_all.str.len() >= 4).sum())

for threshold in [0, 4, 8, 12, 16, 24, 32]:
    keep = text_all.str.len() >= threshold
    rows_kept = int(keep.sum())

    threshold_rows.append(
        {
            "min_chars": threshold,
            "rows_kept": rows_kept,
            "retention_ratio_overall": float(rows_kept / len(raw_reviews)) if len(raw_reviews) else np.nan,
            "retention_vs_non_empty_base": float(rows_kept / non_empty_base) if non_empty_base else np.nan,
            "avg_text_len": float(text_all[keep].str.len().mean()) if rows_kept else np.nan,
        }
    )

threshold_df = pd.DataFrame(threshold_rows)
display(threshold_df)


# %%

fig_threshold = px.line(
    threshold_df,
    x="min_chars",
    y=["retention_ratio_overall", "retention_vs_non_empty_base"],
    markers=True,
    title="Retention by text-length threshold (overall and vs non-empty baseline)",
    height=420,
)
fig_threshold.show()

# Data-driven threshold choice (unless manual override is provided in cfg.min_chars)
if cfg.min_chars is not None:
    selected_min_chars = int(cfg.min_chars)
    selection_reason = "manual override from cfg.min_chars"
else:
    # Exclude min_chars < 4 from automatic selection to avoid keeping empty/micro text.
    candidates = threshold_df[threshold_df["min_chars"] >= 4]
    eligible = candidates[candidates["retention_vs_non_empty_base"] >= cfg.min_retention_target]

    if eligible.empty:
        selected_min_chars = int(candidates.loc[candidates["retention_vs_non_empty_base"].idxmax(), "min_chars"])
        selection_reason = (
            f"fallback: no threshold reached target retention {cfg.min_retention_target:.0%} "
            "relative to non-empty baseline"
        )
    else:
        selected_min_chars = int(eligible["min_chars"].max())
        selection_reason = (
            f"largest threshold that keeps at least {cfg.min_retention_target:.0%} "
            "of non-empty baseline reviews"
        )

print(f"Selected min_chars = {selected_min_chars} ({selection_reason})")

df = raw_reviews[["review_text"]].copy()

# Apply transformations
df["clean_no_stem"] = prep_no_stem.transform_series(df["review_text"])
df["clean_stem"] = prep_stem.transform_series(df["review_text"])

# Vocabulary report
vocab_report = pd.DataFrame(
    {
        "raw_text": text_stats(df["review_text"]),
        "clean_no_stem": text_stats(df["clean_no_stem"]),
        "clean_stem": text_stats(df["clean_stem"]),
    }
).T.reset_index(names="view")

# Reduction metric
vocab_no_stem = vocab_report.loc[vocab_report.view == "clean_no_stem", "vocab_size"].iloc[0]
vocab_stem = vocab_report.loc[vocab_report.view == "clean_stem", "vocab_size"].iloc[0]
vocab_reduction_pct = 100 * (vocab_no_stem - vocab_stem) / max(vocab_no_stem, 1)

# %%
# --- Collision analysis (stemming is expected to be available) ---
if prep_stem.stemmer is None:
    raise RuntimeError("Stemming was requested but no Portuguese stemmer is available.")

tokens = (
    df["review_text"]
    .str.lower()
    .str.findall(r"[a-zà-öø-ÿ]+")
    .explode()
    .dropna()
)
tokens = tokens[tokens.str.len() >= 2]

unique_tokens = tokens.drop_duplicates()
stems = unique_tokens.map(prep_stem.stemmer.stem)
mapping = pd.DataFrame({"token": unique_tokens, "stem": stems})

grouped = mapping.groupby("stem")["token"].agg(list)
collisions = grouped[grouped.str.len() > 1]
collision_ratio = len(collisions) / max(len(grouped), 1)

collision_df = (
    collisions.sort_values(key=lambda x: x.str.len(), ascending=False)
    .head(15)
    .apply(
        lambda toks: {
            "n_original_tokens": len(toks),
            "examples": ", ".join(sorted(toks)[:5]),
        }
    )
    .apply(pd.Series)
    .reset_index()
)

# --- Summary ---
summary = pd.DataFrame(
    {
        "metric": [
            "vocab_reduction_pct_stem_vs_no_stem",
            "stem_collision_ratio",
            "n_stem_collisions",
        ],
        "value": [
            round(vocab_reduction_pct, 2),
            round(collision_ratio, 4),
            int(len(collisions)),

# Build topic naming/table artifacts for one model.
def build_topic_outputs(model_key: str, Xm: np.ndarray, lm: np.ndarray):
    exemplars_m = topic_modeler.select_exemplars(
        texts=reviews["review_text"],
        embeddings=Xm,
        labels=lm,
    )
    keyword_names_m = topic_modeler.keyword_topic_names(
        texts=reviews["review_text"],
        labels=lm,
    )

    topic_name_map_m = keyword_names_m
    topic_labels_m = pd.Series(lm).map(topic_name_map_m).fillna("unlabeled").to_numpy()

    topic_table_m = topic_modeler.build_topic_table(
        texts=reviews["review_text"],
        embeddings=Xm,
        labels=lm,
        sentiment_labels=reviews["sentiment_label"],
    )
    topic_table_m["topic_name_keyword"] = topic_table_m["topic_name"]
    topic_table_m["topic_name"] = topic_table_m["cluster_id"].map(topic_name_map_m).fillna(topic_table_m["topic_name_keyword"])
    topic_table_m["model_key"] = model_key

    return exemplars_m, topic_name_map_m, topic_labels_m, topic_table_m

def build_negative_outputs(model_key: str, Xm: np.ndarray, lm: np.ndarray, topic_labels_m: np.ndarray):

    disp_topic_m = dispersion_profile(Xm, topic_labels_m)
    disp_topic_m["model_key"] = model_key
    disp_topic_m["label_set"] = "topic"

    neg_profile_m = evaluator.negative_profile(
        embeddings=Xm,
        labels=lm,
        sentiments=reviews["sentiment_label"],
    ).copy()
    neg_profile_m["model_key"] = model_key

    global_neg_m = float(neg_binary.mean())
    se = np.sqrt(np.clip((global_neg_m * (1.0 - global_neg_m)) / neg_profile_m["size"].to_numpy(), 1e-12, None))
    neg_profile_m["z_negative"] = (neg_profile_m["negative_ratio"].to_numpy() - global_neg_m) / se

    density_cut_m = float(neg_profile_m["dispersion"].quantile(NEG_ISLAND_DENSITY_Q))
    neg_profile_m["dense_flag"] = neg_profile_m["dispersion"] <= density_cut_m
    neg_profile_m["island_flag"] = (neg_profile_m["z_negative"] >= NEG_ISLAND_Z_MIN) & neg_profile_m["dense_flag"]

    island_count_m = int(neg_profile_m["island_flag"].sum())
    island_mass_m = float(neg_profile_m.loc[neg_profile_m["island_flag"], "size"].sum() / len(reviews))

    hom_share_m, _, hom_excess_m = negative_homophily(Xm, sent_arr, k=NEG_HOMOPHILY_K)

    boot_m = evaluator.bootstrap_profile(
        labels=lm,
        sentiments=reviews["sentiment_label"],
        n_bootstrap=400,
        confidence=0.95,
    )
    boot_m["model_key"] = model_key

    return disp_sent_m, disp_topic_m, neg_profile_m, boot_m, global_neg_m, hom_share_m, hom_excess_m, island_count_m, island_mass_m

# Wrapper to assemble all RQ metrics for one model.
def compute_phase4_for_model(model_key: str) -> dict:
    Xm = phase3_by_model[model_key]["embeddings"]
    lm = phase3_by_model[model_key]["labels"].astype(int)

    exemplars_m, topic_name_map_m, topic_labels_m, topic_table_m = build_topic_outputs(model_key, Xm, lm)
    (
        disp_sent_m,
        disp_topic_m,
        neg_profile_m,
        boot_m,
        global_neg_m,
        hom_share_m,
        hom_excess_m,
        island_count_m,
        island_mass_m,
    ) = build_negative_outputs(model_key, Xm, lm, topic_labels_m)

    kmeans_row = phase3_by_model[model_key]["purity_table"].query("algorithm == 'kmeans'").iloc[0]
    cluster_neg_std_m = float(topic_table_m["negative_ratio"].std())
    rq_signal_m = "islands" if (cluster_neg_std_m >= 0.12 and island_count_m > 0) else "diffuse"

    row = {
        "model_key": model_key,
        "n_clusters": int(topic_table_m["cluster_id"].nunique()),
        "global_negative_ratio": global_neg_m,
        "cluster_negative_ratio_std": cluster_neg_std_m,
        "negative_homophily_k20": hom_share_m,
        "negative_homophily_excess_k20": hom_excess_m,
        "negative_island_count": island_count_m,
        "negative_island_mass": island_mass_m,
        "topic_dispersion_mean": float(disp_topic_m["mean_pairwise_cosine_distance"].mean()),
        "sentiment_dispersion_mean": float(disp_sent_m["mean_pairwise_cosine_distance"].mean()),
        "kmeans_purity_topic_proxy": float(kmeans_row["purity_topic_proxy"]),
        "kmeans_purity_sentiment": float(kmeans_row["purity_sentiment"]),
        "rq_signal": rq_signal_m,
    }

    return {
        "exemplars": exemplars_m,
        "topic_name_map": topic_name_map_m,
        "topic_labels": topic_labels_m,
        "topic_table": topic_table_m,
        "disp_sent": disp_sent_m,
        "disp_topic": disp_topic_m,
        "negative_profile": neg_profile_m,
        "bootstrap_ci": boot_m,
        "global_neg_ratio": global_neg_m,
        "cluster_neg_std": cluster_neg_std_m,
        "homophily_share": hom_share_m,
        "homophily_excess": hom_excess_m,
        "island_count": island_count_m,
        "island_mass": island_mass_m,
        "rq_signal": rq_signal_m,
        "phase4_row": row,
    }

# --- RQ2: pairwise bootstrap deltas for cluster_negative_ratio_std ---
rng = np.random.default_rng(cfg.random_state)

alpha = 1.0 - BOOT_CONF

    lm = phase3_by_model[model_key]["labels"].astype(int)
    boot_vals = bootstrap_cluster_neg_std(lm, neg_binary, boot_indices)
    cluster_neg_std_boot[model_key] = boot_vals
    cluster_neg_std_boot_rows.append(
        {
            "model_key": model_key,
            "metric": "cluster_negative_ratio_std",
            "estimate": float(boot_vals.mean()),
            "ci_low": float(np.quantile(boot_vals, alpha / 2.0)),
            "ci_high": float(np.quantile(boot_vals, 1.0 - alpha / 2.0)),
            "n_bootstrap": int(N_BOOTSTRAP_RQ2),
            "confidence": float(BOOT_CONF),
        }
    )

cluster_neg_std_bootstrap_ci = pd.DataFrame(cluster_neg_std_boot_rows).sort_values("model_key").reset_index(drop=True)

pairwise_rows = []
for model_a, model_b in combinations(model_run_order, 2):
    delta = cluster_neg_std_boot[model_a] - cluster_neg_std_boot[model_b]
    ci_low = float(np.quantile(delta, alpha / 2.0))
    ci_high = float(np.quantile(delta, 1.0 - alpha / 2.0))

    if ci_low > 0:
        stronger = model_a
    elif ci_high < 0:
        stronger = model_b
    else:
        stronger = "inconclusive"

    pairwise_rows.append(
        {
            "model_a": model_a,
            "model_b": model_b,
            "metric": "cluster_negative_ratio_std",
            "delta_mean": float(delta.mean()),
            "ci_low": ci_low,
            "ci_high": ci_high,
            "stronger_islandization": stronger,
            "n_bootstrap": int(N_BOOTSTRAP_RQ2),
            "confidence": float(BOOT_CONF),
        }
    )

rq2_pairwise_delta_ci = pd.DataFrame(pairwise_rows).sort_values(["model_a", "model_b"]).reset_index(drop=True)
